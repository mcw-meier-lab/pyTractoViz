506 function getMediaType(filePath) {{ 

507 if (!filePath) return null; 

508 // Check if it's a numeric score (for shape_similarity_score) 

509 if (typeof filePath === 'number' || (!isNaN(parseFloat(filePath)) && isFinite(filePath))) {{ 

510 return 'score'; 

511 }} 

512 const ext = filePath.split('.').pop().toLowerCase(); 

513 if (['gif'].includes(ext)) return 'gif'; 

514 if (['mp4', 'webm', 'ogg'].includes(ext)) return 'video'; 

515 if (['png', 'jpg', 'jpeg', 'svg', 'webp'].includes(ext)) return 'image'; 

516 return 'image';
